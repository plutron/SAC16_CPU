.data
    counter: .word 5
    sum:     .word 0
.text
START:
    LDI R1, counter
    LD  R2, R1         ; R2 = counter = 5
    LDI R3, sum
    LD  R4, R3         ; R4 = sum = 0

SUM_LOOP:
    MOV ACC, R4
    ADD R2               ; ACC = sum + counter
    MOV R4, ACC          ; R4 = new sum
    ST  R4, R3         ; sum = R4 (update memory)

    LDI R5, 1
    MOV ACC, R2
    SUB R5               ; ACC = counter - 1
    MOV R2, ACC
    ST  R2, R1         ; counter--

    MOV ACC, R2
    JNZ SUM_LOOP

    HALT
